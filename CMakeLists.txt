cmake_minimum_required(VERSION 3.20...4.1)
project(UseTYPST)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}")
include(UseTYPST)
add_typst_document(diagram.typ DEFAULT_FORMATS svg EXCLUDE_FROM_ALL)
cmake_path(SET image_output_dir "${CMAKE_BINARY_DIR}")
cmake_path(RELATIVE_PATH image_output_dir BASE_DIRECTORY "${CMAKE_SOURCE_DIR}")
# prepend '/' to relative path of image output directory so typst evaluates it relative to project root
add_typst_document(example.typ KEY_VALUE_PAIRS image_output_dir=/${image_output_dir} DEPENDS diagram)
set(SLIDE_FORMAT presentation-4-3)
add_typst_document(slides.typ TARGET slides-short PAGES 1-3 KEY_VALUE_PAIRS format=${SLIDE_FORMAT})
add_typst_document(slides.typ TARGET slides-full KEY_VALUE_PAIRS format=${SLIDE_FORMAT})
add_subdirectory(subdir)
add_subdirectory(SampleProject)

option(USETYPST_BUILD_TESTS "Build the UseTYPST integration tests" ON)
if(USETYPST_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
