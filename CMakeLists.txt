cmake_minimum_required(VERSION 3.20...4.1)
project(UseTYPST)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}")
include(UseTYPST)
add_typst_document(DOCUMENT diagram ROOT_FILE diagram.typ OUTPUT_FORMAT svg)
add_typst_document(DOCUMENT example ROOT_FILE example.typ DEPENDS diagram)
set(SLIDE_FORMAT presentation-4-3)
add_typst_document(DOCUMENT slides-short ROOT_FILE slides.typ PAGES 1-3 KEY_VALUE_PAIRS format=${SLIDE_FORMAT})
add_typst_document(DOCUMENT slides-full ROOT_FILE slides.typ KEY_VALUE_PAIRS format=${SLIDE_FORMAT})
add_subdirectory(subdir)
add_subdirectory(SampleProject)
